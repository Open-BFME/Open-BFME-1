static inline Bool isAreaClear(Object *object, const ReplenishUnitsBehaviorModuleData *data)
{
	if (data->m_blockRadius == g_bfmeDefaultBR)
		return true;

	Rva001DCBB0Filter playerFilter(object, false);
	Rva0025ED50ObjectFilter sameMap(object);
	Rva0025ED50RootFilter root;
	PartitionFilterRelationship relationship(object, 1, false);
	PartitionFilter *filters = relationship.link(root.link(sameMap.link(&playerFilter)));
	return ThePartitionManager->bfmeGo1050D((Int)&object->m_position,
		*(const Int *)&data->m_blockRadius, 0, (Int)filters) == 0;
}

// ?update@ReplenishUnitsBehavior@@UAE?AW4UpdateSleepTime@@XZ
UpdateSleepTime ReplenishUnitsBehavior::update()
{
	ReplenishUnitsBehaviorPrimary *self = this;
	const ReplenishUnitsBehaviorModuleData *data = m_moduleData;
	Object *object = m_object;

	if (!isActive() || (object->m_privateStatus & 1))
		return UPDATE_SLEEP_FOREVER;

	void *contain = 0;
	if (data->m_needsContain)
	{
		if (!object->m_contain || !(contain = object->m_contain->slot26()))
			return UPDATE_SLEEP_FOREVER;
	}

	if (isAreaClear(object, data))
	{
		Rva00204360VptrZeroObject containFilter(contain, true);
		Rva0025ED50ObjectFilter sameMap(object);
		Rva0025ED50RootFilter root;
		VptrZeroBlock12 first = { 0, 0x20000, 0 };
		Rva001FC280VptrZeroBlockObject kindOf(first, data->m_kindOf);
		Rva0016A9F0FunctorValueWrapper player(object->getControllingPlayer());
		PartitionFilter *filters = player.link(kindOf.link(root.link(sameMap.link(&containFilter))));
		void *found = ThePartitionManager->bfmeGo1050D((Int)&object->m_position,
			*(const Int *)&data->m_replenishRadius, 0, (Int)filters);
		if (found)
			((BfmeHostEQW *)self)->bfmeGoEQW((BfmeThingEQW *)found);
	}
	return (UpdateSleepTime)data->m_sleep88;
}
