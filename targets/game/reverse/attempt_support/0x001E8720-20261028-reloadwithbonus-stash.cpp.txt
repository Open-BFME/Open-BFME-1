enum WeaponSlotType
{
	PRIMARY_WEAPON = 0,
	SECONDARY_WEAPON,
	TERTIARY_WEAPON,
	// Retail fans out over four slots (cmp edi,4); no fourth EA name is
	// established, so the count is spelled without inventing one.
	WEAPONSLOT_COUNT = 4
};

enum WeaponStatus
{
	READY_TO_FIRE,
	OUT_OF_AMMO,
	BETWEEN_FIRING_SHOTS,
	RELOADING_CLIP,
	PRE_ATTACK
};

class Weapon;

// WeaponSet layout per retail: m_curWeaponTemplateSet at +4 (Snapshot vtable
// at +0), m_weapons at +8. isSharedReloadTime is the 17-byte body at
// 0x001EB140 (pinned); getWeaponInWeaponSlot is the 11-byte body at
// 0x001EAF90 (rowed, reached via ILT 0x0003C8E9).
class WeaponSet
{
public:
	Weapon *getWeaponInWeaponSlot(WeaponSlotType wslot) const;
	Bool isSharedReloadTime() const;

	const void *m_snapshotBase;
	const void *m_curWeaponTemplateSet;
	Weapon *m_weapons[WEAPONSLOT_COUNT];
};

class Object
{
public:
	BfmeAmmoProvider *getAmmoProvider() const { return m_contain; }
	Bool isReloadTimeShared() const { return m_weaponSet.isSharedReloadTime(); }
	Weapon *getWeaponInWeaponSlot(WeaponSlotType wslot) const { return m_weaponSet.getWeaponInWeaponSlot(wslot); }

	char m_pad0[0x1fc];
	BfmeAmmoProvider *m_contain;
	char m_pad200[0x264 - 0x200];
	WeaponSet m_weaponSet;
};

// Retail's global at 0x012F0898 is EA's `GameLogic *TheGameLogic`, defined once

class GameLogic
{
public:
	Object *findObjectByID(ObjectID id);
	UnsignedInt getFrame() { return m_frame; }

	char m_pad[0x3c];
	UnsignedInt m_frame;

extern GameLogic *TheGameLogic;

// m_clipSize at +0x4AC, ammo descriptor at +0x4E8 (same placement as landed
// Weapon_getRemainingAmmo.cpp). getClipReloadTime is the rowed 107-byte body
// at 0x001E1680.
class WeaponTemplate
{
public:
	Int getClipSize() const { return m_clipSize; }
	Int getClipReloadTime(const WeaponBonus &bonus) const;

	char m_pad[0x4ac];
	Int m_clipSize;
	BfmeHostERT m_ammoDescriptor;
};

class Weapon
{
public:
	virtual void unused();
	void setPossibleNextShotFrame(UnsignedInt frame) { m_whenWeCanFireAgain = frame; }
	WeaponStatus getStatus() const { return m_status; }

protected:
	void reloadWithBonus(const Object *sourceObj, const WeaponBonus &bonus, Bool loadInstantly);
	void rebuildScatterTargets();
	void setStatus(WeaponStatus status) { m_status = status; }

	WeaponTemplate *m_template;
	ObjectID m_ownerID;
	UnsignedInt m_weaponSlot;
	WeaponStatus m_status;
	UnsignedInt m_ammoInClip;
	UnsignedInt m_whenWeCanFireAgain;
	char m_pad1C[0x28 - 0x1c];
	UnsignedInt m_whenLastReloadStarted;
};

void Weapon::reloadWithBonus(const Object *sourceObj, const WeaponBonus &bonus, Bool loadInstantly)
{
	if (m_template->getClipSize() > 0)
	{
		UnsignedInt remaining;
		if (m_template->m_ammoDescriptor.bfmeQueryERT())
		{
			Object *owner = TheGameLogic->findObjectByID(m_ownerID);
			BfmeAmmoProvider *provider = owner ? owner->getAmmoProvider() : 0;
			if (provider)
				remaining = provider->getRemainingAmmo(&m_template->m_ammoDescriptor);
			else
				remaining = m_ammoInClip;
		}
		else
			remaining = m_ammoInClip;
		if (remaining == m_template->getClipSize()
			&& !sourceObj->isReloadTimeShared())
			return;	// don't restart our reload delay.
	}

	m_ammoInClip = m_template->getClipSize();
	UnsignedInt providerAmmo;
	if (m_template->m_ammoDescriptor.bfmeQueryERT())
	{
		Object *owner = TheGameLogic->findObjectByID(m_ownerID);
		BfmeAmmoProvider *provider = owner ? owner->getAmmoProvider() : 0;
		if (provider)
			providerAmmo = provider->getRemainingAmmo(&m_template->m_ammoDescriptor);
		else
			providerAmmo = m_ammoInClip;
	}
	else
		providerAmmo = m_ammoInClip;
	if (providerAmmo <= 0)
		m_ammoInClip = 0x7fffffff;	// 0 == unlimited (or effectively so)

	if (m_status != RELOADING_CLIP)
		m_status = RELOADING_CLIP;

	Int reloadTime;
	if (loadInstantly)
		reloadTime = 0;
	else
		reloadTime = m_template->getClipReloadTime(bonus);

	m_whenWeCanFireAgain = (UnsignedInt)((double)reloadTime + (m_whenLastReloadStarted = TheGameLogic->getFrame()));

	// If we are sharing reload times, go through the other weapons in the
	// weapon set: set their m_whenWeCanFireAgain to this guy's delay and
	// their m_status to this guy's status.
	if (sourceObj->isReloadTimeShared())
	{
		for (Int wt = 0; wt < WEAPONSLOT_COUNT; wt++)
		{
			Weapon *weapon = sourceObj->getWeaponInWeaponSlot((WeaponSlotType)wt);
			if (weapon)
			{
				weapon->setPossibleNextShotFrame(m_whenWeCanFireAgain);
				if (weapon->getStatus() != RELOADING_CLIP)
					weapon->setStatus(RELOADING_CLIP);
			}
		}
	}

	rebuildScatterTargets();
}
