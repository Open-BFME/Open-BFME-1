#include <math.h>
typedef float Real;
struct Coord3D
{
	Real x, y, z;
	Coord3D() {}
	Coord3D(const Coord3D &other) { x = other.x; y = other.y; z = other.z; }
	Real length() const { return (Real)sqrt(z * z + y * y + x * x); }
	void normalize()
	{
		Real len = length();
		if (len != 0.0f)
		{
			Real inv = 1.0f / len;
			x *= inv;
			y *= inv;
			z *= inv;
		}
	}
	void sub(const Coord3D *a) { x -= a->x; y -= a->y; z -= a->z; }
	void scale(Real s) { x *= s; y *= s; z *= s; }
};
