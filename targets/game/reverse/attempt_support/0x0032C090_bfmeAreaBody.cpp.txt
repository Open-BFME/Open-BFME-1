	Parameter *pTriggerParm)
{

	PolygonTrigger *pTrig = TheScriptEngine->getQualifiedTriggerAreaByName(pTriggerParm->m_string);
	if (pTrig == 0)
		return false;

	UnsignedShort playerMask = TheScriptEngine->unidentified_0034DB40(pPlayerParm);
	while (playerMask)
	{

		if (!pPlayer)
			continue;

		if (pCondition->getCustomData() != 0 &&
			TheScriptEngine->getFrameObjectCountChanged() <= pCondition->getCustomFrame() &&
			TheScriptEngine->getFrameRva170DC() <= pCondition->getCustomFrame())
		{
			if (pCondition->getCustomData() == -1)
				continue;
			if (pCondition->getCustomData() == 1)
				return true;
		}

		_ReadWriteBarrier();

		for (const PlayerTeamNode *it = pPlayer->getPlayerTeams()->begin();
			it != pPlayer->getPlayerTeams()->end(); it = it->m_next)
		{
			for (BfmeTeamIterator titer = it->m_data->iterate_TeamInstanceList(); !titer.done(); titer.advance())

					continue;

		}
