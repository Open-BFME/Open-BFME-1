    ~BFMERetailAsciiString() { reinterpret_cast<AsciiString *>(this)->~AsciiString(); }
    int compare(const BfmeSharedString &other) const
    {
        return reinterpret_cast<const AsciiString *>(this)->compare(reinterpret_cast<const AsciiString &>(other));
    }
private:
    unsigned char storage[sizeof(AsciiString)];
};

__forceinline BFMERetailAsciiString::BFMERetailAsciiString(Rva00361960 *producer)
{
    union {
        BfmeSharedString (Rva00361960::*original)();
        void (Rva00361960::*fill)(BFMERetailAsciiString *);
    } call;
    call.original = &Rva00361960::copyString;
    (producer->*call.fill)(this);
}

class Rva00363360LivingWorldEntryStore
{
public:
	Rva00361960 *findByField10(const BfmeSharedString &key);

private:
	char m_beforeEntries[0x18];
	std::vector<Rva00361960> m_entries;
};

Rva00361960 *Rva00363360LivingWorldEntryStore::findByField10(
	const BfmeSharedString &key)
{
	for (unsigned int i = 0; i < m_entries.size(); ++i)
	{
		BFMERetailAsciiString current(&m_entries[i]);
		if (current.compare(key) == 0)
			return &m_entries[i];
	}
	return 0;
}
